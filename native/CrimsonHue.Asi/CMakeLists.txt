cmake_minimum_required(VERSION 3.25)
project(CrimsonHueAsi LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "CrimsonHue requires Windows x64.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT EXISTS "${DOTNET_HOST_INCLUDE}/hostfxr.h")
    message(FATAL_ERROR "Pass DOTNET_HOST_INCLUDE from the .NET win-x64 host pack.")
endif()
if(NOT EXISTS "${PAYLOAD_CAB}" OR NOT EXISTS "${PAYLOAD_MANIFEST}")
    message(FATAL_ERROR "Build the managed component cabinet and manifest first.")
endif()
file(TO_CMAKE_PATH "${PAYLOAD_CAB}" CAB_PATH)
file(TO_CMAKE_PATH "${PAYLOAD_MANIFEST}" MANIFEST_PATH)
configure_file(payload.rc.in "${CMAKE_CURRENT_BINARY_DIR}/payload.rc" @ONLY)
add_library(CrimsonHue SHARED host.cpp payload.cpp "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
set_target_properties(CrimsonHue PROPERTIES PREFIX "" SUFFIX ".asi")
target_include_directories(CrimsonHue PRIVATE "${DOTNET_HOST_INCLUDE}")
target_compile_features(CrimsonHue PRIVATE cxx_std_20)
target_compile_options(CrimsonHue PRIVATE /W4 /WX /permissive-)
target_compile_definitions(CrimsonHue PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX NETHOST_USE_AS_STATIC)
target_link_libraries(CrimsonHue PRIVATE "${DOTNET_HOST_INCLUDE}/libnethost.lib" setupapi shell32 ole32 bcrypt)

# Developer-only native loader; never shipped in the ASI package.
add_executable(CrimsonHue.AsiSmoke ../../tests/CrimsonHue.AsiSmoke/main.cpp)
target_compile_features(CrimsonHue.AsiSmoke PRIVATE cxx_std_20)
target_compile_options(CrimsonHue.AsiSmoke PRIVATE /W4 /WX /permissive-)
target_compile_definitions(CrimsonHue.AsiSmoke PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
