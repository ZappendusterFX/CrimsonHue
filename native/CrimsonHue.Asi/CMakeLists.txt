include(FetchContent)
function(hue_dependency name archive url hash)
    set(location "${url}")
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../artifacts/dependency-downloads/${archive}")
        set(location "${CMAKE_CURRENT_SOURCE_DIR}/../../artifacts/dependency-downloads/${archive}")
    endif()
    FetchContent_Declare(${name} URL "${location}" URL_HASH "SHA256=${hash}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
endfunction()

    file(READ "${imgui_SOURCE_DIR}/${header}" stb_source)
    string(FIND "${stb_source}" "ALTERNATIVE A - MIT License" license_start REVERSE)
    if(license_start LESS 0)
        message(FATAL_ERROR "Missing embedded stb license")
    endif()
    string(SUBSTRING "${stb_source}" ${license_start} -1 stb_license)
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/NATIVE-THIRD-PARTY-NOTICES.txt" "${header}\n${stb_license}\n")
endforeach()
